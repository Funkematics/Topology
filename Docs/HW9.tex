\problem Sutherland 10.6

Suppose that $(A, \mathcal{T}_A)$ is a subspace of a topological space $(X, \mathcal{T})$ and let $W \subseteq A$.
\begin{itemize}
  \item[(a)] If $W$ is open in $A$ (that is, if $W \in \mathcal{T}_A$) and $A$ is open in $X$ (that is, if $A \in \mathcal{T}$) then $W$ is open in $X$ (that is, $W \in \mathcal{T}$).
  \item[(b)] If $W$ is closed in $A$ and $A$ is closed in $X$ then $W$ is closed in $X$.
\end{itemize}

\begin{proof}
\textbf{(a)} Since $W$ is open in $A$, there is an open set $U$ in $X$ such that $W = U \cap A$. Since $W$ is an intersection of two open sets in $X$, it is open in $X$.

\textbf{(b)} Since $W$ is closed in $A$, there is a closed set $F$ in $X$ such that $W = F \cap A$. Since $W$ is an intersection of two closed sets in $X$, it is closed in $X$.
\end{proof}

\problem Sutherland 10.7(b) Hint: Prop 9.5

Suppose that $f : X \to Y$ is a map of topological spaces. Prove that $f$ is continuous in each of the following cases.
\begin{itemize}
  \item[(b)] $X = \bigcup_{i=1}^{n} V_i$ where each $V_i$ is a closed subset of $X$ and $f|V_i$ is continuous for each $i$.
\end{itemize}
[Hint: Recall Exercise 3.13.]

\begin{proof}
Let $F \subseteq Y$ be closed. Since the $V_i$ cover $X$,
\[
f^{-1}(F) = \bigcup_{i=1}^{n} \big(f^{-1}(F) \cap V_i\big) = \bigcup_{i=1}^{n} (f|V_i)^{-1}(F).
\]
Each $(f|V_i)^{-1}(F)$ is closed in $V_i$, since $f|V_i$ is continuous, and it is closed in $X$ by 10.6(b). A finite union of closed sets is closed, so $f^{-1}(F)$ is closed. By Prop 9.5, $f$ is continuous.
\end{proof}

\problem Sutherland 10.10

Suppose that $X$, $Y$ are spaces with subspaces $A \subseteq X$, $B \subseteq Y$, and that $f : X \to Y$ is a homeomorphism with $f(A) = B$. Prove that the maps $g : A \to B$ and $h : X \setminus A \to Y \setminus B$ induced by $f$ are both homeomorphisms.

\begin{proof}
Since $f(A) = B$ and $f$ is injective, $g = f|_A : A \to B$ is a bijection with inverse $f^{-1}|_B$. Since $g$ and its inverse are restrictions of continuous maps and their values lie in $B$ and $A$ respectively, $g$ and its inverse are continuous. Hence $g$ is a homeomorphism.

Similarly, since $f$ is a bijection, $f(X \setminus A) = Y \setminus B$. The same argument shows that $h = f|_{X \setminus A}$ is a homeomorphism.
\end{proof}

\problem Sutherland 10.11

Suppose that $X$, $Y$ are topological spaces each with the discrete topology. Prove that the product topology for $X \times Y$ is discrete.

\begin{proof}
In a discrete space, every set is open. So for $(x,y) \in X \times Y$, the sets $\{x\}$ and $\{y\}$ are open, so $\{(x,y)\} = \{x\} \times \{y\}$ is open. Every set in $X \times Y$ is a union of singletons. Suppose $S \subseteq X \times Y$; then $S = \bigcup_{p \in S} \{p\}$, so every $S$ is open.
\end{proof}

\problem Sutherland 10.15 (The first sentence should start ``Prove that \textbf{if}'')

\begin{itemize}
  \item[(a)] Prove that $W$ is open in a topological product $X \times Y$ then $p_X(W)$ is open in $X$ and $p_Y(W)$ is open in $Y$.
  \item[(b)] Give an example of a closed set $W \subseteq \Reals \times \Reals$ whose projection $p_1(W)$ on the $x$-axis is not closed in $\Reals$.
\end{itemize}

\begin{proof}
\textbf{(a)} Let $x \in p_X(W)$, so $(x,y) \in W$ for some $y$. Since $W$ is open in the product topology, there are open sets $U$ and $V$ such that $x \in U$ and $y \in V$ with $U \times V \subseteq W$. For $u \in U$, the point $(u,y)$ is in $U \times V \subseteq W$, so $u \in p_X(W)$. Thus $x \in U \subseteq p_X(W)$, and $p_X(W)$ is open. $p_Y(W)$ is open in $Y$ through a similar argument.

\textbf{(b)} Let $W = \{(x,y) : xy = 1\} = m^{-1}(\{1\})$, where $m(x,y) = xy$ is continuous. Since $\{1\}$ is closed in $\Reals$, $W$ is closed. But
\[
p_1(W) = \Reals \setminus \{0\},
\]
since for $x \neq 0$ the point $(x, 1/x)$ lies in $W$, and no point of $W$ has $x = 0$. This set is not closed, since its complement $\{0\}$ is not open.
\end{proof}

\problem Sutherland 10.16 (i) and (ii) \textbf{(AI Free)}

Suppose that $X$, $Y$ are spaces and that $A \subseteq X$, $B \subseteq Y$. Prove that
\begin{itemize}
  \item[(i)] the interior of $A \times B$ is $\mathring{A} \times \mathring{B}$,
  \item[(ii)] $\overline{A \times B} = \overline{A} \times \overline{B}$.
\end{itemize}

\begin{proof}
\textbf{(i)} We want to show that $\operatorname{int}(A \times B) = \mathring{A} \times \mathring{B}$.

($\supseteq$) $\mathring{A}$ and $\mathring{B}$ are open, so $\mathring{A} \times \mathring{B}$ is open, and it lies in $A \times B$. Since the interior of a set is the largest open set contained in it, $\mathring{A} \times \mathring{B} \subseteq \operatorname{int}(A \times B)$.

($\subseteq$) Consider $(x,y) \in \operatorname{int}(A \times B)$. Since $\operatorname{int}(A \times B)$ is open in the product topology, let $U \times V$ be a basic open set with $(x,y) \in U \times V \subseteq A \times B$. For any $u \in U$, the point $(u,y)$ lies in $A \times B$, so $U \subseteq A$. Similarly $V \subseteq B$. Since $U$ is open with $x \in U \subseteq A$, and $V$ is open with $y \in V \subseteq B$, it follows that $x \in \mathring{A}$ and $y \in \mathring{B}$.

\textbf{(ii)} We show $\overline{A \times B} = \overline{A} \times \overline{B}$ by proving both directions.

($\subseteq$) $\overline{A} \times \overline{B}$ is closed, since its complement $\big((X \setminus \overline{A}) \times Y\big) \cup \big(X \times (Y \setminus \overline{B})\big)$ is open. It also contains $A \times B$. Since $\overline{A \times B}$ is the smallest closed set containing $A \times B$, $\overline{A \times B} \subseteq \overline{A} \times \overline{B}$.

($\supseteq$) A point lies in the closure of a set iff every basic open set containing it meets the set. Let $x \in \overline{A}$, $y \in \overline{B}$, and let $(x,y) \in U \times V$ with $U$, $V$ open. Since $x \in \overline{A}$ and $U$ is an open set containing $x$, we can pick $a \in U \cap A$. Similarly pick $b \in V \cap B$. Then $(a,b) \in (U \times V) \cap (A \times B)$. So every basic open set containing $(x,y)$ meets $A \times B$, and $(x,y) \in \overline{A \times B}$.
\end{proof}

\problem Sutherland 10.17

Suppose that $X \times X$ is the topological product of a space $X$ with itself. Prove that the map $t : X \times X \to X \times X$ defined by $t(x, x') = (x', x)$ is a homeomorphism.

\begin{proof}
Since $t(t(x,x')) = (x,x')$, we have $t \circ t = \mathrm{id}$, so $t$ is a bijection with $t^{-1} = t$. The coordinates of $t$ are the projections $p_2$ and $p_1$, which are continuous, and a map into a product is continuous iff its coordinates are, so $t$ is continuous. Since $t^{-1} = t$, its inverse is also continuous. Thus $t$ is a homeomorphism.
\end{proof}
